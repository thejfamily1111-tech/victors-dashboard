def get_live_premarket_indices():
    """Fetches real-time quotes for SPY and QQQ."""
    data = {}
    for sym in ["SPY", "QQQ"]:
        try:
            ticker_obj = yf.Ticker(sym)
            df = ticker_obj.history(period="2d", interval="1m", prepost=True)
            if not df.empty:
                prev_close = ticker_obj.info.get("previousClose") or df["Close"].iloc[0]
                current_price = float(df["Close"].iloc[-1])
                pct = ((current_price - prev_close) / prev_close) * 100.0
                data[sym] = {"price": current_price, "pct": pct}
            else:
                data[sym] = {"price": 0.0, "pct": 0.0}
        except Exception:
            data[sym] = {"price": 0.0, "pct": 0.0}
    return data


def fetch_htf_macro_confluence(symbol: str):
    """Pulls completed 1-Hour and 4-Hour bars to audit structural trend."""
    try:
        df_1h, df_4h, ts_1h, ts_4h = mb.fetch_completed_htf_bars(symbol)
        if df_1h.empty or len(df_1h) < 15:
            return None

        c_1h = float(df_1h["Close"].iloc[-1])
        ema_1h = float(df_1h["EMA20"].iloc[-1])
        prev_ema_1h = float(df_1h["EMA20"].iloc[-3]) if len(df_1h) >= 3 else ema_1h

        slope_1h = "Bullish ↗" if ema_1h >= prev_ema_1h else "Bearish ↘"
        trend_1h = "🟢 Above 20 EMA (Bullish)" if c_1h >= ema_1h else "🔴 Below 20 EMA (Bearish)"

        if not df_4h.empty and len(df_4h) >= 5:
            c_4h = float(df_4h["Close"].iloc[-1])
            ema_4h = float(df_4h["EMA20"].iloc[-1])
            res_4h = float(df_4h["High"].iloc[-15:].max())
            sup_4h = float(df_4h["Low"].iloc[-15:].min())
            air_to_res = ((res_4h - c_4h) / c_4h) * 100.0
            trend_4h = "🟢 Above 20 EMA" if c_4h >= ema_4h else "🔴 Below 20 EMA"
        else:
            c_4h = c_1h
            ema_4h = ema_1h
            res_4h = float(df_1h["High"].max())
            sup_4h = float(df_1h["Low"].min())
            air_to_res = 1.0
            trend_4h = "🟢 Neutral"

        if c_1h >= ema_1h and air_to_res >= 1.0:
            vic_status = "🟢 PERMIT (Clear Air)"
        elif c_1h >= ema_1h and air_to_res < 1.0:
            vic_status = "🟡 DOWNGRADE (Near 4H Pivot)"
        else:
            vic_status = "🔴 VETO (Counter-Trend Drag)"

        return {
            "1H_Close": f"${c_1h:.2f}",
            "1H_EMA20": f"${ema_1h:.2f}",
            "1H_Trend": trend_1h,
            "1H_Slope": slope_1h,
            "4H_Close": f"${c_4h:.2f}",
            "4H_EMA20": f"${ema_4h:.2f}",
            "4H_Trend": trend_4h,
            "4H_Key_Res": f"${res_4h:.2f} (+{air_to_res:.2f}%)",
            "4H_Key_Sup": f"${sup_4h:.2f}",
            "VIC_Status": vic_status,
        }
    except Exception:
        return None

COLOR_GOOD = "#00e676"
COLOR_NEUTRAL = "#ffb300"
COLOR_BAD = "#ff5252"


def fmt_score_card(label: str, val_text: str, badge_text: str, status: str) -> str:
    color = COLOR_GOOD if status == "good" else (COLOR_NEUTRAL if status == "neutral" else COLOR_BAD)
    return f"""
    <div class="metric-badge-card">
        <div class="metric-badge-label">{label}</div>
        <div class="metric-badge-val" style="color: {color};">{val_text}</div>
        <div style="font-size: 0.78rem; font-weight: 600; color: {color};">{badge_text}</div>
    </div>
    """


def fmt_spread_str(spread_pct: float) -> str:
    if spread_pct <= 4.0:
        return f'<span style="color: {COLOR_GOOD}; font-weight: bold;">{spread_pct:.2f}% (Liquid)</span>'
    elif spread_pct <= 6.0:
        return f'<span style="color: {COLOR_NEUTRAL}; font-weight: bold;">{spread_pct:.2f}% (Borderline)</span>'
    return f'<span style="color: {COLOR_BAD}; font-weight: bold;">{spread_pct:.2f}% ⚠️ (TOO WIDE)</span>'


def fmt_delta_str(delta: float) -> str:
    abs_d = abs(delta)
    if 0.60 <= abs_d <= 0.75:
        return f'<span style="color: {COLOR_GOOD}; font-weight: bold;">{abs_d:.2f} (Target Band: 0.60–0.75)</span>'
    elif (0.50 <= abs_d < 0.60) or (0.75 < abs_d <= 0.82):
        return f'<span style="color: {COLOR_NEUTRAL}; font-weight: bold;">{abs_d:.2f} (Acceptable Drift)</span>'
    return f'<span style="color: {COLOR_BAD}; font-weight: bold;">{abs_d:.2f} (Suboptimal Delta)</span>'


def fmt_rs_str(rs_val: float) -> str:
    if rs_val > 0.10:
        return f'<span style="color: {COLOR_GOOD}; font-weight: bold;">+{rs_val:.2f}% (Leading)</span>'
    elif rs_val >= -0.10:
        return f'<span style="color: {COLOR_NEUTRAL}; font-weight: bold;">{rs_val:+.2f}% (In-Line)</span>'
    return f'<span style="color: {COLOR_BAD}; font-weight: bold;">{rs_val:.2f}% (Lagging Drag)</span>'
